            public void FixedUpdate()
            {
                float xDeltaMove = Input.GetAxis("Horizontal");
                float zDeltaMove = Input.GetAxis("Vertical");

                Vector3 movement = new Vector3(xDeltaMove, 0, zDeltaMove);

                //直接根据输入量的大小的方向，制定我机的速度
                this.GetComponent<Rigidbody>().velocity = movement * MoveSpeed;

                //根据输入量的大小，来确定我机的倾斜角度
                this.GetComponent<Rigidbody>().rotation = Quaternion.Euler(
                    0.0f,
                    0.0f,
                    this.GetComponent<Rigidbody>().velocity.x * -tilt  //根据我机速度的x分量
                    );
            }
